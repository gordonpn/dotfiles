# -------- instant prompt (must stay first) --------
# Draws the starship prompt before the rest of this bundle and .zshrc load,
# then swaps in the real one. `post:` runs the starship init, which is why
# .zshrc no longer calls `eval "$(starship init zsh)"` itself.
#
# ftl-transient collapses finished commands to the `transient` profile in
# starship.toml. It needs zsh 5.9+: it calls `zle widget -f nolast` when another
# plugin already owns zle-line-finish (fast-syntax-highlighting does), and that
# form was only added in 5.9. On 5.8.1 and older it errors after every command
# with `unknown option: f`, so the guard skips it there.
mattmc3/starship-ftl post:"ftl-prompt starship; autoload -Uz is-at-least; is-at-least 5.9 && ftl-transient on"

# OMZ dependency glue (recommended if using ohmyzsh/ohmyzsh path:...)
getantidote/use-omz

# OMZ base (load before OMZ plugins)
# Cherry-picked OMZ Libs (Faster than loading all of path:lib)
ohmyzsh/ohmyzsh path:lib/directories.zsh
ohmyzsh/ohmyzsh path:lib/key-bindings.zsh

# 1. Completions First (Blocking but fast if static)
greymd/docker-zsh-completion kind:fpath
$HOME/.zsh/plugins/uv-completions kind:fpath

# 2. Core Functionality (Load immediately)
MenkeTechnologies/zsh-expand

# 3. Deferred Plugins (Load after prompt is visible)

# Autosuggestions is async by default but deferring init helps the first draw
zsh-users/zsh-autosuggestions kind:defer

# Fast Syntax Highlighting is already optimized, but deferring ensures prompt is instant
zdharma-continuum/fast-syntax-highlighting kind:defer

# 4. Interactive Tools (Lazy Load)
# Use 'zsh-bd' only when you type 'bd'

# 5. History Search (Must load AFTER syntax highlighting to avoid conflicts)
zsh-users/zsh-history-substring-search kind:defer

# 6. FZF Tab (Heavy, so defer it)
